/** * TEAM KEI 🔥 — Backend de datos para el dashboard * Pégalo en Extensiones → Apps Script del Sheet "Team Kei🔥". * * v3: * - Alias CR -> cr_prom_pct (tu pestaña KPIs_Mensuales tiene el encabezado * "CR", no "cr_prom_pct" — el dashboard esperaba ese segundo nombre). * - Los referidos de equipo ahora se leen automáticamente de la pestaña * que tenga encabezados MES / META / LOGRADO (sin importar su nombre * ni si tiene una fila de título arriba, como "TEAM ELITE"). */ const CACHE_KEY = 'team_kei_dashboard_v3'; const CACHE_SECONDS = 3 * 60 * 60; // 3 horas function doGet(e) { const cache = CacheService.getScriptCache(); const forceRefresh = e && e.parameter && e.parameter.refresh === '1'; if (!forceRefresh) { const cached = cache.get(CACHE_KEY); if (cached) { return ContentService.createTextOutput(cached).setMimeType(ContentService.MimeType.JSON); } } const ss = SpreadsheetApp.getActiveSpreadsheet(); const kpis = sheetToObjects(ss, 'KPIs_Mensuales', ['cumplimiento_pct', 'cr_prom_pct', 'CR']); // Alias: si el encabezado real es "CR", lo copiamos también a cr_prom_pct kpis.forEach(function (k) { if ((k.cr_prom_pct === undefined || k.cr_prom_pct === '' || k.cr_prom_pct === null) && k.CR !== undefined) { k.cr_prom_pct = k.CR; } }); const data = { historico: sheetToObjects(ss, 'Historico_Elite', ['pct_cumplimiento', 'cr_pct']), kpis: kpis, roster: sheetToObjects(ss, 'Roster'), metasReferidos: readMetasReferidos(ss), nuevoModelo: readNuevoModelo(ss), modeloAnterior: readModeloAnterior(ss), generadoEn: new Date().toISOString() }; const json = JSON.stringify(data); try { cache.put(CACHE_KEY, json, CACHE_SECONDS); } catch (err) { // payload > 100KB — se ignora el cache, no es un error fatal } return ContentService.createTextOutput(json).setMimeType(ContentService.MimeType.JSON); } /** * Convierte una pestaña simple (fila 1 = encabezados) en un array de objetos. * percentFields: columnas formateadas como "Porcentaje" en Sheets — para esas, * se lee el texto mostrado ("93%") en vez del valor crudo (0.93). */ function sheetToObjects(ss, name, percentFields) { percentFields = percentFields || []; const sheet = ss.getSheetByName(name); if (!sheet) return []; const range = sheet.getDataRange(); const values = range.getValues(); const displayValues = range.getDisplayValues(); if (values.length < 2) return []; const headers = values[0]; const result = []; for (let r = 1; r < values.length; r++) { const row = values[r]; if (row[0] === '' || row[0] === null) continue; const obj = {}; headers.forEach((h, i) => { if (percentFields.indexOf(h) !== -1) { const disp = String(displayValues[r][i]).replace('%', '').replace(',', '.').trim(); const num = parseFloat(disp); obj[h] = isNaN(num) ? row[i] : num; } else { obj[h] = row[i]; } }); result.push(obj); } return result; } /** * Busca en TODAS las pestañas del Sheet una que tenga una fila con * encabezados MES / META / LOGRADO (sin importar si tiene un título * arriba, como "TEAM ELITE", ni cómo se llame la pestaña). * Devuelve [{mes, meta, logrado}, ...] */ function readMetasReferidos(ss) { const sheets = ss.getSheets(); for (let s = 0; s < sheets.length; s++) { const sheet = sheets[s]; const values = sheet.getDataRange().getValues(); for (let i = 0; i < values.length; i++) { const row = values[i].map(c => String(c).trim().toUpperCase()); if (row[0] === 'MES' && row[1] === 'META') { const result = []; for (let r = i + 1; r < values.length; r++) { const mes = String(values[r][0] || '').trim(); if (!mes || mes.toUpperCase().indexOf('TOTAL') === 0) continue; result.push({ mes: mes, meta: values[r][1], logrado: values[r][2] }); } return result; } } } return []; } /** NUEVO_MODELO tiene título + tabla de tiers + fijo mensual + reglas de referidos. */ function readNuevoModelo(ss) { const sheet = ss.getSheetByName('NUEVO_MODELO'); if (!sheet) return null; const values = sheet.getDataRange().getValues(); const tiers = []; let fijoMensual = 0, refPrimero = 0, refResto = 0, refCero = -100; for (let i = 0; i < values.length; i++) { const row = values[i]; if (row[0] === 'tier') { for (let j = i + 1; j < values.length; j++) { if (!values[j][0]) break; tiers.push({ tier: values[j][0], pct_comision: values[j][1], pagos_min: values[j][2], nr_min: values[j][3], condicion: values[j][4] }); } } if (row[0] === 'Fijo mensual') fijoMensual = row[1]; if (String(row[0]).indexOf('1er referido') === 0) refPrimero = row[1]; if (String(row[0]).indexOf('2do en adelante') === 0) refResto = row[1]; if (String(row[0]).indexOf('0 referidos') === 0) refCero = row[1]; } return { tiers: tiers, fijoMensual: fijoMensual, referidos: { primero: refPrimero, resto: refResto, cero: refCero } }; } /** MODELO_ANTERIOR tiene título + tabla de bandas de comisión. */ function readModeloAnterior(ss) { const sheet = ss.getSheetByName('MODELO_ANTERIOR'); if (!sheet) return null; const values = sheet.getDataRange().getValues(); const bandas = []; for (let i = 0; i < values.length; i++) { if (values[i][0] === 'pct_cumplimiento_min') { for (let j = i + 1; j < values.length; j++) { if (values[j][0] === '' || values[j][0] === null) break; bandas.push({ min: values[j][0], max: values[j][1], comision_pct: values[j][2] }); } break; } } return { bandas: bandas }; }
